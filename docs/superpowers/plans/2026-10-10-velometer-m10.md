# velometer M10: Polish and Accessibility Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A dark theme that follows the OS, wide tables that stack into cards on phones, and WCAG 2.1 AA conformance across every page of the reader app. Each is proven by rule tests in the gate and by an axe-core render pass of the static edition.

**Architecture:**
- Colours become CSS custom properties only, with a `prefers-color-scheme: dark` block.
- JS reads tokens through one small module (`theme.js`) and redraws when the theme changes.
- One tested colour-maths module (`velometer/report/colour.py`) proves contrast and palette distinctness.
- A reusable dev tool (`tools/render_pass.py` with vendored axe-core) drives headless Chrome over the exported static edition.

**Tech Stack:** Python 3 / FastAPI / Jinja2, plain ES modules (`node --test`), pytest, uv. axe-core and `websocket-client` are dev-only.

**Spec:** `docs/superpowers/specs/2026-10-09-velometer-m10-design.md`

## Global Constraints

- No displayed number, run, classifier or DB change. Never write `velometer.sqlite`; its md5 is unchanged by every task.
- The app's only external host stays `tile.openstreetmap.org`. No npm packages, no JS build step.
- axe-core is never served by the app and never exported. It lives in `tools/axe/` with its MPL-2.0 licence and a VERSION file.
- OSM figures are "mapped km" (ADR 0006). Attribution lines unchanged (ADR 0008).
- Tests assert rules, not today's values. The one equality allowed is the single-source rule: the light tier, outline and change-category tokens equal the Python constants.
- Every fix has a test that fails without it.
- Gate: `uv run ruff format --check . && uv run ruff check . && uv run pytest && node --test "src/velometer/app/static/js/tests/*.test.mjs"`.
- The render pass needs Chrome and stays out of the gate.
- Commits: explicit paths (never `git add -A`), with only the trailer `Assisted-by: Claude:claude-opus-5-5`. Never push.
- Contrast thresholds:
  - 4.5:1 for text;
  - 3:1 for large text and non-text (input borders, focus ring, chart lines, tiers against chart and map backgrounds).
- Palette thresholds (CIEDE2000):
  - tier palette: every pair at least ΔE 20 in normal vision, deuteranopia and protanopia, and adjacent tiers (A–B, B–C, C–D, D–E) at least ΔE 25;
  - series palette (ruling, see Task 2): every pair at least ΔE 20 in normal vision. Series lines are direct-labelled and listed in tables, so colour-blind distinctness is not required of all 8.
- axe tags: `wcag2a`, `wcag2aa`, `wcag21a`, `wcag21aa`, with zero violations.
- Render widths are 1280, 390 and 320 px. Prose tables are the only elements allowed to scroll sideways.

```python
"""Colour maths for the accessibility rules: WCAG contrast, CIELAB, CIEDE2000, colour-blindness
simulation (Machado, Oliveira & Fernandes 2009, severity 1.0). sRGB hex in, numbers out."""

import math
from typing import Literal

_MACHADO = {
    "protanopia": ((0.152286, 1.052583, -0.204868), (0.114503, 0.786281, 0.099216),
                   (-0.003882, -0.048116, 1.051998)),
    "deuteranopia": ((0.367322, 0.860646, -0.227968), (0.280085, 0.672501, 0.047413),
                     (-0.011820, 0.042940, 0.968881)),
}  # fmt: skip


def _linear(hexcolour: str) -> tuple[float, float, float]:
    h = hexcolour.lstrip("#")
    if len(h) == 3:
        h = "".join(c * 2 for c in h)
    srgb = [int(h[i : i + 2], 16) / 255 for i in (0, 2, 4)]
    return tuple(c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4 for c in srgb)


def _hex(linear: tuple[float, float, float]) -> str:
    def enc(c: float) -> int:
        c = min(1.0, max(0.0, c))
        s = c * 12.92 if c <= 0.0031308 else 1.055 * c ** (1 / 2.4) - 0.055
        return round(s * 255)

    return "#" + "".join(f"{enc(c):02x}" for c in linear)


def _luminance(hexcolour: str) -> float:
    r, g, b = _linear(hexcolour)
    return 0.2126 * r + 0.7152 * g + 0.0722 * b


def contrast(a: str, b: str) -> float:
    """WCAG 2.x contrast ratio, 1–21."""
    la, lb = sorted((_luminance(a), _luminance(b)), reverse=True)
    return (la + 0.05) / (lb + 0.05)


def lab(hexcolour: str) -> tuple[float, float, float]:
    """CIELAB (D65) of an sRGB colour."""
    r, g, b = _linear(hexcolour)
    x = (0.4124 * r + 0.3576 * g + 0.1805 * b) / 0.95047
    y = 0.2126 * r + 0.7152 * g + 0.0722 * b
    z = (0.0193 * r + 0.1192 * g + 0.9505 * b) / 1.08883

    def f(t: float) -> float:
        return t ** (1 / 3) if t > 216 / 24389 else (24389 / 27 * t + 16) / 116

    return 116 * f(y) - 16, 500 * (f(x) - f(y)), 200 * (f(y) - f(z))


def ciede2000(lab1, lab2) -> float:
    """CIEDE2000 colour difference (kL = kC = kH = 1)."""
    L1, a1, b1 = lab1
    L2, a2, b2 = lab2
    cbar = (math.hypot(a1, b1) + math.hypot(a2, b2)) / 2
    g = 0.5 * (1 - math.sqrt(cbar**7 / (cbar**7 + 25**7)))
    a1p, a2p = (1 + g) * a1, (1 + g) * a2
    c1p, c2p = math.hypot(a1p, b1), math.hypot(a2p, b2)
    h1p = math.degrees(math.atan2(b1, a1p)) % 360 if (b1 or a1p) else 0.0
    h2p = math.degrees(math.atan2(b2, a2p)) % 360 if (b2 or a2p) else 0.0
    dlp, dcp = L2 - L1, c2p - c1p
    if c1p * c2p == 0:
        dhp = 0.0
    else:
        dhp = h2p - h1p
        if dhp > 180:
            dhp -= 360
        elif dhp < -180:
            dhp += 360
    dHp = 2 * math.sqrt(c1p * c2p) * math.sin(math.radians(dhp / 2))
    lbp, cbp = (L1 + L2) / 2, (c1p + c2p) / 2
    if c1p * c2p == 0:
        hbp = h1p + h2p
    elif abs(h1p - h2p) <= 180:
        hbp = (h1p + h2p) / 2
    elif h1p + h2p < 360:
        hbp = (h1p + h2p + 360) / 2
    else:
        hbp = (h1p + h2p - 360) / 2
    t = (1 - 0.17 * math.cos(math.radians(hbp - 30)) + 0.24 * math.cos(math.radians(2 * hbp))
         + 0.32 * math.cos(math.radians(3 * hbp + 6)) - 0.20 * math.cos(math.radians(4 * hbp - 63)))  # fmt: skip
    dtheta = 30 * math.exp(-(((hbp - 275) / 25) ** 2))
    rc = 2 * math.sqrt(cbp**7 / (cbp**7 + 25**7))
    sl = 1 + 0.015 * (lbp - 50) ** 2 / math.sqrt(20 + (lbp - 50) ** 2)
    sc = 1 + 0.045 * cbp
    sh = 1 + 0.015 * cbp * t
    rt = -math.sin(math.radians(2 * dtheta)) * rc
    return math.sqrt((dlp / sl) ** 2 + (dcp / sc) ** 2 + (dHp / sh) ** 2
                     + rt * (dcp / sc) * (dHp / sh))  # fmt: skip


def simulate(hexcolour: str, kind: Literal["deuteranopia", "protanopia"]) -> str:
    """How the colour looks with full deuteranopia or protanopia."""
    m = _MACHADO[kind]
    rgb = _linear(hexcolour)
    return _hex(tuple(sum(m[i][j] * rgb[j] for j in range(3)) for i in range(3)))


def delta(a: str, b: str, vision: str = "normal") -> float:
    """CIEDE2000 between two colours as seen with `vision` (normal or a simulated deficiency)."""
    if vision != "normal":
        a, b = simulate(a, vision), simulate(b, vision)
    return ciede2000(lab(a), lab(b))
```

import itertools
import re

import pytest

from tests.test_app import _css_outside_tokens, _css_rules, _css_tokens
from velometer.report import change, charts
from velometer.report.colour import contrast

from tests.test_app_m8 import _stylesheet

THEMES = ("light", "dark")
TEXT_PROPS = ("color",)
BG_PROPS = ("background", "background-color")
NONTEXT_PROPS = ("border", "border-color", "border-top", "border-bottom", "border-left",
                 "border-right", "outline", "outline-color", "stroke", "fill", "box-shadow")  # fmt: skip
VAR = re.compile(r"var\((--[\w-]+)\)")


def _used(props) -> set[str]:
    return {
        t
        for _, decls in _css_rules(_css_outside_tokens(_stylesheet()))
        for p, v in decls.items()
        if p in props
        for t in VAR.findall(v)
    }


def _backgrounds() -> set[str]:
    # the page and card grounds every text sits on, plus every background a rule sets
    return _used(BG_PROPS) | {"--paper", "--surface", "--panel"}


@pytest.mark.parametrize("theme", THEMES)
def test_every_text_token_reads_on_every_background(theme):
    tokens = _css_tokens(_stylesheet(), theme)
    bad = [
        (fg, bg, round(contrast(tokens[fg], tokens[bg]), 2))
        for fg, bg in itertools.product(sorted(_used(TEXT_PROPS)), sorted(_backgrounds()))
        if fg != bg and contrast(tokens[fg], tokens[bg]) < 4.5
        and (fg, bg) not in TEXT_EXEMPT
    ]
    assert bad == []

def test_text_exemptions_name_real_tokens():

// Colours come from the stylesheet's tokens (app.css), so charts and maps follow the OS theme.
// `read` is injectable for node tests; in the page it reads the root's computed style.
export const MISSING = "#808080";
const warned = new Set();

const pageRead = (name) =>
  globalThis.getComputedStyle?.(globalThis.document.documentElement).getPropertyValue(name).trim() ?? "";

/** The value of one colour token. A missing token logs one console error (the render pass fails on it). */
export function token(name, read = pageRead) {
  const v = read(name);
  if (v) return v;
  if (!warned.has(name)) {
    warned.add(name);
    console.error(`velometer: colour token ${name} is not defined`);
  }
  return MISSING;
}

/** The 8 series colours, slot 0 first. */
export const seriesColours = (read = pageRead) =>
  Array.from({ length: 8 }, (_, i) => token(`--series-${i + 1}`, read));

/** Tier letter -> colour. */
export const tierColours = (read = pageRead) =>
  Object.fromEntries([..."ABCDE"].map((t) => [t, token(`--tier-${t.toLowerCase()}`, read)]));

/** Call `fn` whenever the OS colour scheme changes; returns an unsubscribe function. */
export function onThemeChange(fn, matchMedia = globalThis.matchMedia) {
  const q = matchMedia?.("(prefers-color-scheme: dark)");
  q?.addEventListener("change", fn);
  return () => q?.removeEventListener("change", fn);
}
