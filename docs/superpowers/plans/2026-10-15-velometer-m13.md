```python
def sum_counters(per_counter: CounterHours) -> list[HourCount]:
    """The site-hours of one file: the sum of the site's counters, kept only when every counter of
    the site in service at that hour reports it. A counter is in service from its first to its last
    reported hour in the file (naive local time, fold breaking the tie of a repeated hour); a
    site-hour one of them lacks inside that span is left out, so to_days marks the day
    incomplete."""
    span: dict[tuple[str, str], tuple[tuple[datetime, int], tuple[datetime, int]]] = {}
    for site, counter, t, fold in per_counter:
        k = (t, fold)
        lo, hi = span.get((site, counter), (k, k))
        span[site, counter] = (min(lo, k), max(hi, k))
    counters: dict[str, list[tuple[tuple[datetime, int], tuple[datetime, int]]]] = defaultdict(list)
    for (site, _), lo_hi in span.items():
        counters[site].append(lo_hi)
    total: dict[tuple[str, datetime, int], int] = defaultdict(int)
    reporting: dict[tuple[str, datetime, int], int] = defaultdict(int)
    for (site, _, t, fold), count in per_counter.items():
        total[site, t, fold] += count
        reporting[site, t, fold] += 1
    out = []
    for (site, t, fold), count in total.items():
        k = (t, fold)
        if reporting[site, t, fold] == sum(lo <= k <= hi for lo, hi in counters[site]):
            out.append(HourCount(site, t.replace(fold=fold), count))
    return out
```

  Note: `(t, fold)` tuples order the repeated autumn hour correctly, because the first 02:00
  (fold 0) sorts before the second (fold 1). Spring-forward has no repeated wall time, so the
  placeholder hour keeps the larger count upstream, as now.
  If profiling shows the per-hour `sum` is slow on the real Paris files (about 10 M rows), cache
  per `(site, t, fold)`; correctness first.

from velometer.counters.index import chain_index, chain_index_v1
from velometer.counters.model import DayCount


def _year(site, year, per_day, skip=0):
    """A site's days in `year`, each counting `per_day`, with the last `skip` days incomplete."""
    first = date(year, 1, 1)
    n = (date(year + 1, 1, 1) - first).days
    return [
        DayCount(site, first + timedelta(i), per_day, i < n - skip) for i in range(n)
    ]


def _sites(year, spec):
    return [d for site, per_day, skip in spec for d in _year(site, year, per_day, skip)]


def test_v2_link_is_unmoved_by_a_coverage_difference_at_equal_daily_volume():
    # same daily volume both years; site c has 30 fewer complete days in 2021 (still >= 90 %)
    days = _sites(2020, [("a", 100, 0), ("b", 200, 0), ("c", 300, 0)])
    days += _sites(2021, [("a", 100, 0), ("b", 200, 0), ("c", 300, 30)])
    v2 = dict((y, i) for y, i, _ in chain_index(days, 2020))
    v1 = dict((y, i) for y, i, _ in chain_index_v1(days, 2020))
    assert v2[2021] == 100.0  # same mean per complete day
    assert v1[2021] < 100.0  # the v1 sum falls with coverage


def test_v2_link_is_the_pooled_mean_ratio():
    days = _sites(2020, [("a", 100, 0), ("b", 100, 0), ("c", 100, 0)])
    days += _sites(2021, [("a", 110, 0), ("b", 110, 0), ("c", 110, 0)])
    assert dict((y, i) for y, i, _ in chain_index(days, 2020))[2021] == 110.0

- [ ] **Step 1: Define "real loss" as a measurement.** It is the dropped pieces (kept by v3,
  dropped by v4) that lie *along* the network across a turn: the sum of
  1. the hole pieces (`--checks` "holes");
  2. the `along` class;
  3. the class-boundary pieces counted in neither class;

     midpoint lies within `buffer` of a network line of the same class, where that line
     continues after a turn of more than 30° within `END_TOL_M` + the buffer. M8 did not
     measure these.

  sum. Put the outer-corner test in a function `outer_corner(piece, lines, params) -> bool`, with
  a unit test on synthetic geometry in `tests/test_overlay_false_positives.py`: an L-shaped
  network line; a piece at the outer corner, offset 8 m, is True; a piece 40 m past a straight
  line's end is False. Load the script with `importlib`, as the other script tests do (grep for
  `importlib` in tests).
